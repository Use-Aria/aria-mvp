    1. Create `src/app/api/auth/google/route.ts` -- GET handler:
    - Read `user_id` from query param `?user_id=...`
    - Generate HMAC state: `crypto.createHmac('sha256', process.env.GOOGLE_CLIENT_SECRET!).update(user_id + Date.now().toString()).digest('hex')`
    - Store `{state, user_id}` in a signed httpOnly cookie named `oauth_state` (JSON stringified, 10-minute maxAge)
    - Build Google OAuth authorization URL:
      - `https://accounts.google.com/o/oauth2/v2/auth`
      - `client_id` = `process.env.GOOGLE_CLIENT_ID`
      - `redirect_uri` = `process.env.GOOGLE_REDIRECT_URI`
      - `response_type` = `code`
      - `scope` = `https://www.googleapis.com/auth/calendar https://www.googleapis.com/auth/gmail.readonly`
      - `access_type` = `offline`
      - `prompt` = `consent` (forces refresh_token on re-auth)
      - `state` = the HMAC state value
    - Return `NextResponse.redirect(authUrl)`

    ```python
    @celery_app.task(name="tasks.encrypt_and_store_tokens", priority=9)
    def encrypt_and_store_tokens(user_id: str, access_token: str, refresh_token: str) -> dict:
        """Encrypt OAuth tokens and store in Supabase. Then trigger downstream setup.
        Called immediately from OAuth callback -- tokens arrive as plaintext args
        and are encrypted before any DB write."""
        from google_auth.tokens import store_google_tokens
        from sms.send import send_sms_safe, _get_user_sms_state

        # Step 1: Encrypt and store tokens (NEVER plaintext in DB)
        store_google_tokens(user_id, access_token, refresh_token)

        # Step 2: Send confirmation SMS
        user_state = _get_user_sms_state(user_id)
        if user_state and user_state.get("phone"):
            send_sms_safe(
                user_id,
                user_state["phone"],
                "Google connected! I'll now include your calendar and watch for deadline emails.",
                is_proactive=True,
            )

        # Step 3: Check if this is first Google connect (no study_preference set yet)
        # If so, send the study preference question (locked decision from CONTEXT.md)
        from ai.db import _get_supabase
        client = _get_supabase()
        user_row = client.table("users").select("preferences_json").eq("id", user_id).execute().data
        prefs = (user_row[0].get("preferences_json") or {}) if user_row else {}
        if not prefs.get("study_preference"):
            # First Google connect -- ask study preference
            if user_state and user_state.get("phone"):
                send_sms_safe(
                    user_id,
                    user_state["phone"],
                    "Quick question: when do you prefer to study -- mornings, afternoons, or evenings? Just reply with one!",
                    is_proactive=True,
                )

        # Step 4: Dispatch downstream tasks (GCal backfill + Gmail watch)
        # These tasks are defined in Plans 03/04; they will be no-ops if not yet registered
        try:
            gcal_sync_user.delay(user_id)
        except Exception:
            pass  # Task not registered yet (Plan 03)
        try:
            gmail_establish_watch_task.delay(user_id)
        except Exception:
            pass  # Task not registered yet (Plan 04)

        return {"user_id": user_id, "stored": True, "study_pref_asked": not prefs.get("study_preference")}
    ```
