    import os
    from datetime import datetime, timezone
    from google.oauth2.credentials import Credentials
    from google.auth.transport.requests import Request
    from googleapiclient.discovery import build

    def build_gmail_service(user_id: str):
        """Build Gmail API service with auto-refreshing credentials.
        Same token refresh pattern as gcal/auth.py."""
        from google_auth.tokens import load_google_tokens, store_google_tokens

        creds_data = load_google_tokens(user_id)
        if creds_data is None:
            raise ValueError(f"No Google integration for user {user_id}")

        creds = Credentials(
            token=creds_data["access_token"],
            refresh_token=creds_data["refresh_token"],
            token_uri="https://oauth2.googleapis.com/token",
            client_id=os.environ["GOOGLE_CLIENT_ID"],
            client_secret=os.environ["GOOGLE_CLIENT_SECRET"],
        )

        if creds.expired or (creds.expiry and (creds.expiry - datetime.now(timezone.utc)).total_seconds() < 300):
            creds.refresh(Request())
            store_google_tokens(user_id, creds.token, creds.refresh_token)

        return build("gmail", "v1", credentials=creds)

    import base64
    import json
    import logging

    logger = logging.getLogger(__name__)

    def decode_pubsub_message(data: str) -> dict:
        """Decode base64url Pub/Sub data field -> {emailAddress, historyId}"""
        padded = data + "=" * (4 - len(data) % 4)
        return json.loads(base64.urlsafe_b64decode(padded).decode())

    def fetch_new_messages(user_id: str, start_history_id: str) -> list[dict]:
        """Fetch messages added since start_history_id.
        Returns list of {subject, sender, snippet} for .edu senders ONLY.
        Full email content is NEVER returned — only metadata."""
        from gmail.auth import build_gmail_service

        service = build_gmail_service(user_id)
        try:
            result = service.users().history().list(
                userId="me",
                startHistoryId=start_history_id,
                historyTypes=["messageAdded"],
                labelId="INBOX",
            ).execute()
        except Exception as e:
            # 404 means historyId is too old — need full re-sync
            logger.error("Gmail history.list failed for %s: %s", user_id, e)
            return []

        messages = []
        for history in result.get("history", []):
            for msg_added in history.get("messagesAdded", []):
                msg_id = msg_added["message"]["id"]
                # Fetch metadata only — NEVER store full body
                msg = service.users().messages().get(
                    userId="me", id=msg_id, format="metadata",
                    metadataHeaders=["From", "Subject"],
                ).execute()

                headers = {h["name"]: h["value"] for h in msg.get("payload", {}).get("headers", [])}
                sender = headers.get("From", "")
                subject = headers.get("Subject", "")
                snippet = msg.get("snippet", "")

                # Filter to .edu senders only (locked decision)
                if ".edu" not in sender.lower():
                    continue

                messages.append({
                    "subject": subject,
                    "sender": sender,
                    "snippet": snippet,
                    "message_id": msg_id,
                })

        return messages

    def get_stored_history_id(user_id: str) -> str | None:
        """Get last processed historyId from integrations table."""
        from ai.db import _get_supabase
        client = _get_supabase()
        result = (
            client.table("integrations")
            .select("history_id")
            .eq("user_id", user_id)
            .eq("platform", "google")
            .execute()
        )
        if result.data and result.data[0].get("history_id"):
            return result.data[0]["history_id"]
        return None

    def update_history_id(user_id: str, history_id: str) -> None:
        """Update last processed historyId in integrations table."""
        from ai.db import _get_supabase
        client = _get_supabase()
        client.table("integrations").update(
            {"history_id": history_id}
        ).eq("user_id", user_id).eq("platform", "google").execute()

    ```python
    import json
    import logging
    import os
    from anthropic import Anthropic
    from rapidfuzz import fuzz

    logger = logging.getLogger(__name__)

    HAIKU_SYSTEM = """You classify professor emails for a college student assistant.
    Output ONLY valid JSON: {"is_deadline_change": bool, "confidence": float}
    Set confidence 0.0-1.0. High confidence (0.8+) if the email explicitly mentions:
    - A new due date, deadline extension, or postponement
    - Assignment or exam cancellation
    - Schedule change for an exam or major assignment
    Low confidence for: general announcements, office hours changes, logistics."""

    def classify_email(subject: str, snippet: str) -> dict:
        """Classify an email using Claude Haiku.
        Returns {"is_deadline_change": bool, "confidence": float}.
        Falls back to {is_deadline_change: False, confidence: 0.0} on any error."""
        try:
            client = Anthropic(api_key=os.environ["ANTHROPIC_API_KEY"])
            user_content = f"Subject: {subject}\n\nEmail preview: {snippet}"
            response = client.messages.create(
                model="claude-haiku-4-5-20251001",
                max_tokens=64,
                system=HAIKU_SYSTEM,
                messages=[{"role": "user", "content": user_content}],
            )
            return json.loads(response.content[0].text)
        except Exception as e:
            logger.error("Email classification failed: %s", e)
            return {"is_deadline_change": False, "confidence": 0.0}

    def find_matching_deadline(email_subject: str, deadlines: list[dict]) -> dict | None:
        """Find best-matching deadline via rapidfuzz token_set_ratio.
        Returns best match if score >= 70, else None.
        Compares against both deadline title and course name."""
        if not deadlines:
            return None

        best_score = 0
        best_match = None
        subject_lower = email_subject.lower()

        for d in deadlines:
            # Match against title
            title_score = fuzz.token_set_ratio(subject_lower, d.get("title", "").lower())
            # Also try title + course combo
            course = d.get("course") or d.get("course_id") or ""
            combo = f"{d.get('title', '')} {course}".lower()
            combo_score = fuzz.token_set_ratio(subject_lower, combo)
            score = max(title_score, combo_score)

            if score > best_score:
                best_score = score
                best_match = d

        return best_match if best_score >= 70 else None
    ```

    import os
    import logging

    logger = logging.getLogger(__name__)

    PUBSUB_TOPIC = None  # Set from env: GMAIL_PUBSUB_TOPIC e.g. "projects/aria-project/topics/aria-gmail-notifications"

    def establish_gmail_watch(user_id: str) -> dict:
        """Establish Gmail watch via users.watch() with Pub/Sub topic.
        Stores initial historyId in integrations.history_id.
        Watch expires in 7 days — must be renewed."""
        from gmail.auth import build_gmail_service
        from gmail.fetch import update_history_id

        topic = os.environ.get("GMAIL_PUBSUB_TOPIC", "")
        if not topic:
            logger.error("GMAIL_PUBSUB_TOPIC not set — cannot establish watch")
            return {}

        service = build_gmail_service(user_id)
        result = service.users().watch(
            userId="me",
            body={
                "topicName": topic,
                "labelIds": ["INBOX"],
            },
        ).execute()

        # Store historyId as starting point for future history.list calls
        history_id = str(result.get("historyId", ""))
        update_history_id(user_id, history_id)

        # Store expiration for renewal tracking
        expiration = result.get("expiration", 0)
        if expiration:
            from gcal.db import store_watch_channel  # reuse for gmail watch tracking
            # We'll use a separate key to distinguish gmail vs gcal watch
            from ai.db import _get_supabase
            from datetime import datetime, timezone
            client = _get_supabase()
            expires_at = datetime.fromtimestamp(int(expiration) / 1000, tz=timezone.utc).isoformat()
            # Store gmail watch expiry in a gmail-specific field or reuse watch_expires_at
            # Since we need to track both gcal and gmail separately, store gmail expiry
            # in the same integrations row but check history_id presence as gmail indicator
            client.table("integrations").update({
                "watch_expires_at": expires_at,
            }).eq("user_id", user_id).eq("platform", "google").execute()

        logger.info("Gmail watch established for user %s, historyId %s", user_id, history_id)
        return result

    def renew_gmail_watch(user_id: str) -> dict:
        """Renew Gmail watch — simply call watch() again (Gmail handles replacement)."""
        return establish_gmail_watch(user_id)

    def gmail_process_notification(user_id: str, new_history_id: str) -> dict:

        Called when /api/gmail/webhook receives a Pub/Sub push."""
        from gmail.fetch import fetch_new_messages, get_stored_history_id, update_history_id
        from gmail.classify import classify_email, find_matching_deadline
        from ai.db import _get_supabase
        from sms.batch import enqueue_alert
        from sms.send import _get_user_sms_state

        stored_history_id = get_stored_history_id(user_id)
        if not stored_history_id:
            update_history_id(user_id, new_history_id)
            return {"user_id": user_id, "status": "initialized_history_id"}

        messages = fetch_new_messages(user_id, stored_history_id)
        update_history_id(user_id, new_history_id)

        if not messages:
            return {"user_id": user_id, "processed": 0}

        # Get user's Canvas deadlines for cross-reference
        client = _get_supabase()

            client.table("deadlines")
            .select("id, title, due_date, course")
            .eq("user_id", user_id)

            .execute()
        ).data

                        f"Check your course announcements."

                if is_first:
                    flush_sms_batch.apply_async(args=[user_id], countdown=600)

            processed += 1

        return {"user_id": user_id, "processed": processed, "alerts_fired": alerts_fired}

    import { NextResponse } from 'next/server'

    export async function POST(request: Request) {
      // Gmail Pub/Sub sends a JSON body with message.data (base64url encoded)
      try {
        const body = await request.json()
        const data = body?.message?.data
        if (!data) {
          return NextResponse.json({ ok: true })
        }

        // Decode base64url -> {emailAddress, historyId}
        const decoded = JSON.parse(
          Buffer.from(data, 'base64url').toString('utf-8')
        )
        const historyId = decoded.historyId
        const emailAddress = decoded.emailAddress

        if (!historyId || !emailAddress) {
          return NextResponse.json({ ok: true })
        }

        // Look up user_id by email address
        // NOTE: We need to find the user associated with this Gmail address.
        // For now, query integrations table for Google-connected users.
        // In production, store the Google email in integrations on OAuth callback.
        const { createClient } = await import('@supabase/supabase-js')
        const supabase = createClient(
          process.env.NEXT_PUBLIC_SUPABASE_URL!,
          process.env.SUPABASE_SERVICE_ROLE_KEY!
        )

        // For MVP, we can match by checking all Google-connected users
        // Better approach: store google_email in integrations during OAuth
        const { data: integrations } = await supabase
          .from('integrations')
          .select('user_id')
          .eq('platform', 'google')

        // For now, dispatch for all Google-connected users
        // TODO: Add google_email column to integrations for precise matching
        if (integrations && integrations.length > 0) {
