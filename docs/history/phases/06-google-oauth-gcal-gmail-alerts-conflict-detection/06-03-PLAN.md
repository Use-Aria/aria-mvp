    import os
    from datetime import datetime, timezone, timedelta
    from google.oauth2.credentials import Credentials
    from google.auth.transport.requests import Request
    from googleapiclient.discovery import build

    def build_gcal_service(user_id: str):
        """Build Google Calendar API service with auto-refreshing credentials.
        Loads tokens from integrations table (decrypted), refreshes if expiring
        within 5 minutes, stores refreshed token back."""
        from google_auth.tokens import load_google_tokens, store_google_tokens

        creds_data = load_google_tokens(user_id)
        if creds_data is None:
            raise ValueError(f"No Google integration for user {user_id}")

        creds = Credentials(
            token=creds_data["access_token"],
            refresh_token=creds_data["refresh_token"],
            token_uri="https://oauth2.googleapis.com/token",
            client_id=os.environ["GOOGLE_CLIENT_ID"],
            client_secret=os.environ["GOOGLE_CLIENT_SECRET"],
        )

        # Refresh if expired or expiring within 5 minutes
        if creds.expired or (creds.expiry and (creds.expiry - datetime.now(timezone.utc)).total_seconds() < 300):
            creds.refresh(Request())
            store_google_tokens(user_id, creds.token, creds.refresh_token)

        return build("calendar", "v3", credentials=creds)
    ```

    from datetime import datetime, timezone, timedelta

    def fetch_gcal_events(user_id: str, days: int = 7) -> list[dict]:
        """Fetch GCal events for 7-day rolling window. Filters declined + transparent.
        Returns list of dicts: {google_event_id, title, start_time, end_time, type}"""
        from gcal.auth import build_gcal_service

        service = build_gcal_service(user_id)
        now = datetime.now(timezone.utc)
        time_max = now + timedelta(days=days)

        events_result = service.events().list(
            calendarId="primary",
            timeMin=now.isoformat(),
            timeMax=time_max.isoformat(),
            singleEvents=True,
            orderBy="startTime",
            showDeleted=False,
        ).execute()

        items = events_result.get("items", [])
        normalized = []
        for e in items:
            # Skip transparent (free) events
            if e.get("transparency") == "transparent":
                continue
            # Skip events user has declined
            attendees = e.get("attendees", [])
            if any(a.get("self") and a.get("responseStatus") == "declined" for a in attendees):
                continue

            # Normalize start/end — handle dateTime and date (all-day) formats
            start_raw = e.get("start", {})
            end_raw = e.get("end", {})
            start_time = start_raw.get("dateTime") or start_raw.get("date")
            end_time = end_raw.get("dateTime") or end_raw.get("date")

            # Infer type from summary keywords
            title = e.get("summary", "Untitled")
            event_type = _infer_type(title)

            normalized.append({
                "google_event_id": e["id"],
                "title": title,
                "start_time": start_time,
                "end_time": end_time,
                "type": event_type,
                "source": "gcal",
            })

        return normalized

    def _infer_type(title: str) -> str:
        """Infer event type from title. Returns 'exam', 'class', 'study', or 'personal'."""
        lower = title.lower()
        if any(kw in lower for kw in ("exam", "midterm", "final", "quiz", "test")):
            return "exam"
        if any(kw in lower for kw in ("lecture", "class", "section", "seminar", "lab")):
            return "class"
        if any(kw in lower for kw in ("study", "review", "homework", "pset")):
            return "study"
        return "personal"
    ```

    from datetime import datetime, timezone

    def store_calendar_events(user_id: str, events: list[dict]) -> int:
        """Upsert GCal events into calendar_events table. Returns count stored."""
        from ai.db import _get_supabase
        client = _get_supabase()
        count = 0
        for e in events:
            payload = {
                "user_id": user_id,
                "google_event_id": e["google_event_id"],
                "title": e["title"],
                "start_time": e["start_time"],
                "end_time": e["end_time"],
                "type": e["type"],
            }
            client.table("calendar_events").upsert(
                payload, on_conflict="user_id,google_event_id"
            ).execute()
            count += 1
        return count

    def get_calendar_events(user_id: str, start_date: str, end_date: str) -> list[dict]:
        """Fetch calendar_events for user within date range."""
        from ai.db import _get_supabase
        client = _get_supabase()
        result = (
            client.table("calendar_events")
            .select("id, google_event_id, title, start_time, end_time, type")
            .eq("user_id", user_id)
            .gte("start_time", start_date)
            .lte("start_time", end_date)
            .order("start_time", desc=False)
            .execute()
        )
        return result.data

    def store_watch_channel(user_id: str, channel_id: str, resource_id: str, expiration_ms: int) -> None:
        """Store GCal watch channel info in integrations table."""
        from ai.db import _get_supabase
        from datetime import datetime, timezone
        client = _get_supabase()
        expires_at = datetime.fromtimestamp(expiration_ms / 1000, tz=timezone.utc).isoformat()
        client.table("integrations").update({
            "channel_id": channel_id,
            "resource_id": resource_id,
            "watch_expires_at": expires_at,
        }).eq("user_id", user_id).eq("platform", "google").execute()

    def get_watch_channel(user_id: str) -> dict | None:
        """Get watch channel state from integrations table. Returns dict or None."""
        from ai.db import _get_supabase
        client = _get_supabase()
        result = (
            client.table("integrations")
            .select("channel_id, resource_id, watch_expires_at")
            .eq("user_id", user_id)
            .eq("platform", "google")
            .execute()
        )
        if result.data and result.data[0].get("channel_id"):
            return result.data[0]
        return None
    ```

    import os
    import uuid
    import logging
    from datetime import datetime, timezone, timedelta

    logger = logging.getLogger(__name__)

    def establish_watch(user_id: str) -> dict:
        """Register a GCal push notification channel for user's primary calendar.
        Stores channel state in integrations table. Returns watch result dict."""
        from gcal.auth import build_gcal_service
        from gcal.db import store_watch_channel

        service = build_gcal_service(user_id)
        channel_id = str(uuid.uuid4())
        expiration = int((datetime.now(timezone.utc) + timedelta(days=7)).timestamp() * 1000)

        body = {
            "id": channel_id,
            "type": "web_hook",
            "address": f"{os.environ['APP_BASE_URL']}/api/gcal/webhook",
            "token": user_id,  # returned in X-Goog-Channel-Token header
            "expiration": expiration,
        }
        result = service.events().watch(calendarId="primary", body=body).execute()
        store_watch_channel(user_id, result["id"], result["resourceId"], result["expiration"])
        logger.info("GCal watch established for user %s, channel %s, expires %s",
                     user_id, result["id"], result["expiration"])
        return result

    def stop_watch(user_id: str, channel_id: str, resource_id: str) -> None:
        """Stop an existing GCal push notification channel."""
        from gcal.auth import build_gcal_service
        service = build_gcal_service(user_id)
        service.channels().stop(body={"id": channel_id, "resourceId": resource_id}).execute()
        logger.info("GCal watch stopped: channel %s", channel_id)

    def renew_watch(user_id: str) -> dict | None:
        """Renew a GCal watch channel: stop old, establish new. Returns new watch result or None."""
        from gcal.db import get_watch_channel
        existing = get_watch_channel(user_id)
        if existing and existing.get("channel_id"):
            try:
                stop_watch(user_id, existing["channel_id"], existing["resource_id"])
            except Exception as e:
                logger.warning("Failed to stop old watch for %s: %s", user_id, e)
        return establish_watch(user_id)
    ```

    @celery_app.task(name="tasks.renew_watch_channels")
    def renew_watch_channels():
        """Check all Google integrations for expiring watch channels.
        Renew any channel expiring within 48 hours."""
        from ai.db import _get_supabase
        from datetime import datetime, timezone, timedelta

        client = _get_supabase()
        cutoff = (datetime.now(timezone.utc) + timedelta(hours=48)).isoformat()
        rows = (
            client.table("integrations")
            .select("user_id, channel_id, watch_expires_at")
            .eq("platform", "google")
            .not_.is_("channel_id", "null")
            .lte("watch_expires_at", cutoff)
            .execute()
        ).data

        renewed = 0
        for row in rows:
            try:
                from gcal.watch import renew_watch
                renew_watch(row["user_id"])
                renewed += 1
            except Exception as e:
                import logging
                logging.getLogger(__name__).error("Watch renewal failed for %s: %s", row["user_id"], e)

        return {"checked": len(rows), "renewed": renewed}
