    1. Create `celery_worker/sms/batch.py`:

    ```python
    import json
    import os
    import logging
    import redis

    logger = logging.getLogger(__name__)

    BATCH_TTL = 600  # 10 minutes in seconds
    MAX_TEXTS_PER_WINDOW = 3  # SMS-11

    def _get_redis():
        return redis.from_url(os.environ["REDIS_URL"])

    def enqueue_alert(user_id: str, text: str, priority_score: int) -> bool:
        """Add alert to user's hold queue.
        Returns True if this is the first item (caller should schedule flush_sms_batch.apply_async(args=[user_id], countdown=BATCH_TTL)).
        Uses SETNX for atomicity to prevent duplicate flush scheduling."""
        r = _get_redis()
        key = f"sms_batch:{user_id}"
        lock_key = f"sms_batch_lock:{user_id}"
        item = json.dumps({"text": text, "priority": priority_score})
        r.rpush(key, item)
        r.expire(key, BATCH_TTL)
        # Atomic lock: only first enqueuer gets True
        is_new = r.set(lock_key, "1", nx=True, ex=BATCH_TTL)
        return bool(is_new)

    def flush_batch(user_id: str) -> str | None:
        """Pop all items from queue, sort by priority desc, send one combined SMS.
        Enforces SMS-11: max 3 items per batch. Returns sent body or None."""
        r = _get_redis()
        key = f"sms_batch:{user_id}"
        lock_key = f"sms_batch_lock:{user_id}"

        # Atomically read and delete
        raw_items = r.lrange(key, 0, -1)
        r.delete(key)
        r.delete(lock_key)

        if not raw_items:
            return None

        items = sorted(
            [json.loads(i) for i in raw_items],
            key=lambda x: x["priority"],
            reverse=True,
        )

        # SMS-11: cap at MAX_TEXTS_PER_WINDOW
        items = items[:MAX_TEXTS_PER_WINDOW]

        if len(items) == 1:
            body = items[0]["text"]
        else:
            lines = "\n".join(f"{i+1}) {item['text']}" for i, item in enumerate(items))
            body = f"{len(items)} updates:\n{lines}"

        # Send via send_sms_alert (bypasses quiet hours for high-priority)
        from sms.send import send_sms_alert, _get_user_sms_state
        user_state = _get_user_sms_state(user_id)
        if user_state:
            send_sms_alert(user_id, user_state["phone"], body)
            return body

        logger.warning("flush_batch: no user state for %s", user_id)
        return None
    ```
