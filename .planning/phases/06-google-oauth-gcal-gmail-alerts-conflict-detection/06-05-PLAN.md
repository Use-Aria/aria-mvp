    ```python
    from datetime import datetime, timedelta, time, timezone
    from zoneinfo import ZoneInfo

    WAKING_START = time(7, 0)   # 7am
    WAKING_END = time(22, 0)    # 10pm

    PREFERENCE_ORDER = {
        "morning": [(7, 12), (12, 17), (17, 22)],    # morning, afternoon, evening
        "afternoon": [(12, 17), (7, 12), (17, 22)],
        "evening": [(17, 22), (12, 17), (7, 12)],     # default
    }

    def find_free_slots(
        events: list[dict],
        date: datetime,
        min_duration_hours: int = 2,
        study_preference: str = "evening",
        user_tz: str = "America/New_York",
    ) -> list[dict]:
        """Find free time slots of at least min_duration_hours in waking hours.
        Returns list of {start: datetime, end: datetime, duration_hours: float}
        sorted by preference (study_preference ordering).

        events: list of dicts with start_time and end_time (ISO strings)
        date: the date to search for slots
        """
        tz = ZoneInfo(user_tz)
        day_start = datetime.combine(date.date(), WAKING_START, tzinfo=tz)
        day_end = datetime.combine(date.date(), WAKING_END, tzinfo=tz)

        # Parse and sort events by start time
        busy = []
        for e in events:
            s = datetime.fromisoformat(e["start_time"])
            en = datetime.fromisoformat(e["end_time"])
            if s.tzinfo is None:
                s = s.replace(tzinfo=timezone.utc)
            if en.tzinfo is None:
                en = en.replace(tzinfo=timezone.utc)
            # Convert to user tz
            s = s.astimezone(tz)
            en = en.astimezone(tz)
            # Only include events that overlap with this day's waking hours
            if en > day_start and s < day_end:
                busy.append((max(s, day_start), min(en, day_end)))

        busy.sort(key=lambda x: x[0])

        # Find gaps
        gaps = []
        cursor = day_start
        for start, end in busy:
            if start > cursor:
                gap_hours = (start - cursor).total_seconds() / 3600
                if gap_hours >= min_duration_hours:
                    gaps.append({
                        "start": cursor.isoformat(),
                        "end": start.isoformat(),
                        "duration_hours": round(gap_hours, 1),
                    })
            cursor = max(cursor, end)

        # Check gap after last event
        if cursor < day_end:
            gap_hours = (day_end - cursor).total_seconds() / 3600
            if gap_hours >= min_duration_hours:
                gaps.append({
                    "start": cursor.isoformat(),
                    "end": day_end.isoformat(),
                    "duration_hours": round(gap_hours, 1),
                })

        # Sort by preference
        pref_order = PREFERENCE_ORDER.get(study_preference, PREFERENCE_ORDER["evening"])

        def preference_key(slot):
            slot_hour = datetime.fromisoformat(slot["start"]).hour
            for i, (lo, hi) in enumerate(pref_order):
                if lo <= slot_hour < hi:
                    return i
            return len(pref_order)

        gaps.sort(key=preference_key)
        return gaps
    ```

    ```python
    import logging
    from datetime import datetime, timezone

    logger = logging.getLogger(__name__)

    def detect_conflicts_user(user_id: str) -> list[dict]:
        """Find Canvas deadlines whose due_date (calendar day) overlaps
        any busy GCal event. Deduplicate via conflict_alerts table.
        Returns list of newly fired conflicts."""
        from ai.db import _get_supabase
        from sms.batch import enqueue_alert

        client = _get_supabase()

        # Get active Canvas deadlines
        deadlines = (
            client.table("deadlines")
            .select("id, title, due_date, course")
            .eq("user_id", user_id)
            .is_("deleted_at", "null")
            .neq("status", "completed")
            .execute()
        ).data

        # Get GCal events
        events = (
            client.table("calendar_events")
            .select("id, google_event_id, title, start_time, end_time")
            .eq("user_id", user_id)
            .execute()
        ).data

        if not deadlines or not events:
            return []

        # Get existing conflict alerts for deduplication
        existing = (
            client.table("conflict_alerts")
            .select("deadline_id, gcal_event_id, deadline_due_at")
            .eq("user_id", user_id)
            .execute()
        ).data
        existing_set = {
            (e["deadline_id"], e["gcal_event_id"], e.get("deadline_due_at"))
            for e in existing
        }

        new_conflicts = []
        for d in deadlines:
            d_date_str = d["due_date"][:10]  # YYYY-MM-DD

            for e in events:
                e_date_str = e["start_time"][:10]

                if d_date_str == e_date_str:
                    # Check deduplication: same pair with same deadline date = skip
                    dedup_key = (d["id"], e["google_event_id"], d["due_date"])
                    if dedup_key in existing_set:
                        continue

                    # Calculate event duration for SMS
                    try:
                        e_start = datetime.fromisoformat(e["start_time"])
                        e_end = datetime.fromisoformat(e["end_time"])
                        duration_hrs = round((e_end - e_start).total_seconds() / 3600, 1)
                        duration_str = f"{duration_hrs}-hour" if duration_hrs != 1 else "1-hour"
                    except Exception:
                        duration_str = ""

                    # Build conflict SMS (per CONTEXT.md format)
                    day_name = datetime.fromisoformat(d["due_date"]).strftime("%A")
                    course = d.get("course") or ""
                    body = (
                        f"Heads up: {d['title']}"
                        f"{' (' + course + ')' if course else ''}"
                        f" is due {day_name}, and you have "
                        f"{e['title']} ({duration_str}) that day. "
                        f"Want me to add a study block the day before?"
                    )

                    # Record in conflict_alerts
                    client.table("conflict_alerts").insert({
                        "user_id": user_id,
                        "deadline_id": d["id"],
                        "gcal_event_id": e["google_event_id"],
                        "deadline_due_at": d["due_date"],
                        "gcal_event_date": e_date_str,
                    }).execute()

                    # Route through batch queue
                    is_first = enqueue_alert(user_id, body, priority_score=75)
                    if is_first:
                        from tasks import flush_sms_batch
                        flush_sms_batch.apply_async(args=[user_id], countdown=600)

                    new_conflicts.append({"deadline": d["title"], "event": e["title"]})

        return new_conflicts
    ```

    ```python
    import logging
    from datetime import datetime, timedelta, timezone
    from zoneinfo import ZoneInfo

    logger = logging.getLogger(__name__)

    def create_study_block(
        user_id: str,
        title: str,
        start_dt: datetime,
        end_dt: datetime,
    ) -> str:
        """Create a study block event on user's Google Calendar.
        Returns the Google Calendar event ID (for undo support).

        Args:
            user_id: UUID of the user
            title: Event title (e.g., "Bio Lab Study")
            start_dt: UTC datetime for event start
            end_dt: UTC datetime for event end

        Returns:
            Google Calendar event ID string
        """
        from gcal.auth import build_gcal_service

        service = build_gcal_service(user_id)
        event = {
            "summary": title,
            "start": {"dateTime": start_dt.isoformat(), "timeZone": "UTC"},
            "end": {"dateTime": end_dt.isoformat(), "timeZone": "UTC"},
            "description": "Created by Aria — your AI study assistant",
        }
        created = service.events().insert(calendarId="primary", body=event).execute()
        event_id = created["id"]
        logger.info("Study block created for %s: %s (%s)", user_id, title, event_id)
        return event_id

    def delete_study_block(user_id: str, event_id: str) -> bool:
        """Delete a study block from Google Calendar (undo support).
        Returns True if deleted, False on error."""
        from gcal.auth import build_gcal_service

        try:
            service = build_gcal_service(user_id)
            service.events().delete(calendarId="primary", eventId=event_id).execute()
            logger.info("Study block deleted for %s: %s", user_id, event_id)
            return True
        except Exception as e:
            logger.error("Failed to delete study block %s: %s", event_id, e)
            return False

    def schedule_study_block(
        user_id: str,
        title: str,
        duration_hours: int,
        target_date: datetime,
        study_preference: str = "evening",
        user_tz: str = "America/New_York",
    ) -> dict | None:
        """Find a free slot and create a study block.
        Returns {event_id, start, end, title} or None if no slot available.

        Uses find_free_slots to locate gaps, creates event in first available slot
        matching the user's preference.
        """
        from gcal.db import get_calendar_events
        from gcal.slots import find_free_slots

        date_str = target_date.strftime("%Y-%m-%dT00:00:00+00:00")
        end_str = target_date.strftime("%Y-%m-%dT23:59:59+00:00")
        events = get_calendar_events(user_id, date_str, end_str)

        slots = find_free_slots(
            events=events,
            date=target_date,
            min_duration_hours=duration_hours,
            study_preference=study_preference,
            user_tz=user_tz,
        )

        if not slots:
            return None

        # Use the first (preferred) slot
        slot = slots[0]
        start_dt = datetime.fromisoformat(slot["start"])
        end_dt = start_dt + timedelta(hours=duration_hours)

        event_id = create_study_block(user_id, title, start_dt, end_dt)
        return {
            "event_id": event_id,
            "start": start_dt.isoformat(),
            "end": end_dt.isoformat(),
            "title": title,
        }
    ```

      ```python
      def get_briefing_gcal_context(user_id: str) -> dict:
          """Get GCal enrichment data for Monday briefing.
          Returns {conflicts: list[str], study_windows: list[str]} or empty dict."""
          from ai.db import _get_supabase
          from gcal.db import get_calendar_events
          from gcal.slots import find_free_slots
          from datetime import datetime, timedelta, timezone

          client = _get_supabase()

          # Check if user has Google connected
          integration = (
              client.table("integrations")
              .select("user_id")
              .eq("user_id", user_id)
              .eq("platform", "google")
              .execute()
          ).data
          if not integration:
              return {}

          now = datetime.now(timezone.utc)
          week_end = now + timedelta(days=7)

          # Get conflicts for the week
          conflicts = (
              client.table("conflict_alerts")
              .select("deadline_id, gcal_event_id, deadline_due_at, gcal_event_date")
              .eq("user_id", user_id)
              .gte("gcal_event_date", now.strftime("%Y-%m-%d"))
              .lte("gcal_event_date", week_end.strftime("%Y-%m-%d"))
              .execute()
          ).data

          conflict_strs = []
          for c in conflicts:

              dl = client.table("deadlines").select("title, course").eq("id", c["deadline_id"]).execute().data

              if dl and ev:

                  )

          # Get user preferences
          user = client.table("users").select("preferences_json, timezone").eq("id", user_id).execute().data
          study_pref = "evening"
          user_tz = "America/New_York"
          if user:
              prefs = user[0].get("preferences_json") or {}
              study_pref = prefs.get("study_preference", "evening")
              user_tz = user[0].get("timezone") or "America/New_York"

          # Find best study windows across the week (top 3)
          events = get_calendar_events(
              user_id,
              now.strftime("%Y-%m-%dT00:00:00+00:00"),
              week_end.strftime("%Y-%m-%dT23:59:59+00:00"),
          )

          all_slots = []
          for day_offset in range(7):
              day = now + timedelta(days=day_offset)
              day_events = [e for e in events if e["start_time"][:10] == day.strftime("%Y-%m-%d")]
              slots = find_free_slots(day_events, day, min_duration_hours=2, study_preference=study_pref, user_tz=user_tz)
              for s in slots:
                  start = datetime.fromisoformat(s["start"])
                  day_name = start.strftime("%A")
                  time_str = start.strftime("%-I%p").lower()  # e.g., "6pm"
                  end_time = datetime.fromisoformat(s["end"]).strftime("%-I%p").lower()
                  all_slots.append(f"{day_name} {time_str}-{end_time} free ({s['duration_hours']}h)")

          return {
              "conflicts": conflict_strs,
              "study_windows": all_slots[:3],  # top 3
          }
      ```
