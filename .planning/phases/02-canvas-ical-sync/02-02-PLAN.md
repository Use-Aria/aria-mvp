       ```sql
       -- supabase/migrations/002_deadlines_soft_delete.sql
       -- Add soft-delete support to deadlines table for Phase 2 Canvas sync.
       -- Phase 5 uses deleted_at to alert users when a deadline disappears from Canvas.

       -- Soft-delete column: NULL = active, non-NULL = deleted at that timestamp
       ALTER TABLE public.deadlines
         ADD COLUMN IF NOT EXISTS deleted_at TIMESTAMPTZ;

       -- Index for efficient filtering of active (non-deleted) deadlines
       CREATE INDEX IF NOT EXISTS idx_deadlines_active
         ON public.deadlines (user_id, deleted_at)
         WHERE deleted_at IS NULL;

       -- Unique constraint required for supabase-py upsert on_conflict="canvas_uid,user_id"
       -- canvas_uid is NOT globally unique — must always be scoped with user_id
       ALTER TABLE public.deadlines
         ADD CONSTRAINT deadlines_canvas_uid_user_id_key
         UNIQUE (canvas_uid, user_id);
       ```

    2. Create `celery_worker/canvas/db.py` with these four functions using `supabase-py`:

       ```python
       """Supabase read/write operations for Canvas deadlines and integrations."""
       import os
       from datetime import datetime, timezone
       from supabase import create_client

       def _get_supabase():
           """Create Supabase client using service role key (bypasses RLS)."""
           url = os.environ["SUPABASE_URL"]
           key = os.environ["SUPABASE_SERVICE_ROLE_KEY"]
           return create_client(url, key)

       def get_all_canvas_integrations() -> list[dict]:
           """Return all integrations with platform='canvas' that have an ical_url."""
           client = _get_supabase()
           result = client.table("integrations").select("user_id, ical_url").eq("platform", "canvas").not_.is_("ical_url", "null").execute()
           return result.data

       def load_current_deadlines(user_id: str) -> dict:
           """Load active (non-deleted) deadlines for a user, keyed by canvas_uid.

           Returns:
               Dict of {canvas_uid: {"due_date": "...", "title": "...", ...}}
           """
           client = _get_supabase()
           result = client.table("deadlines").select("canvas_uid, title, due_date, course, type").eq("user_id", user_id).is_("deleted_at", "null").execute()
           return {row["canvas_uid"]: row for row in result.data if row.get("canvas_uid")}

       def apply_diff(user_id: str, diff: dict) -> None:
           """Write added/changed deadlines via upsert, soft-delete removed deadlines.

           Args:
               user_id: UUID of the user.
               diff: Output of compute_diff() — {"added": [...], "changed": [...], "removed": [uid, ...]}.
           """
           client = _get_supabase()
           now = datetime.now(timezone.utc).isoformat()

           # Upsert added + changed deadlines
           for event in diff["added"] + diff["changed"]:
               client.table("deadlines").upsert(
                   {
                       "user_id": user_id,
                       "canvas_uid": event["canvas_uid"],
                       "title": event["title"],
                       "due_date": event["due_date"],
                       "course": event.get("course"),
                       "type": event.get("type"),
                       "last_synced": now,
                       "deleted_at": None,  # clear any prior soft-delete on re-add
                   },
                   on_conflict="canvas_uid,user_id",
               ).execute()

           # Soft-delete removed deadlines
           for uid in diff["removed"]:
               client.table("deadlines").update(
                   {"deleted_at": now}
               ).eq("user_id", user_id).eq("canvas_uid", uid).execute()

       def mark_synced(user_id: str) -> None:
           """Update integrations.synced_at for the user's Canvas integration.

           IMPORTANT: Only call this AFTER apply_diff succeeds.
           """
           client = _get_supabase()
           now = datetime.now(timezone.utc).isoformat()
           client.table("integrations").update(
               {"synced_at": now}
           ).eq("user_id", user_id).eq("platform", "canvas").execute()
       ```

       ```python
       import requests
       from celery_app import celery_app
       from datetime import datetime, timezone


       @celery_app.task(name="tasks.ping")
       def ping():
           """Smoke-test task: logs timestamp to console. DB write added in Phase 2 when real users exist."""
           ts = datetime.now(timezone.utc).isoformat()
           print(f"[ping] executed at {ts}")
           return {"status": "ok", "timestamp": ts}


       @celery_app.task(name="tasks.canvas_sync_all")
       def canvas_sync_all():
           """Query all Canvas integrations and dispatch one sync task per user.

           Called by Celery Beat every 60 minutes. Fans out to canvas_sync_user
           for per-user isolation — one user's failure does not block another.
           """
           from canvas.db import get_all_canvas_integrations

           integrations = get_all_canvas_integrations()
           for row in integrations:
               canvas_sync_user.delay(row["user_id"], row["ical_url"])
           return {"status": "ok", "dispatched": len(integrations)}


       @celery_app.task(
           name="tasks.canvas_sync_user",
           bind=True,
           autoretry_for=(requests.HTTPError,),
           retry_backoff=2,
           retry_backoff_max=300,
           retry_jitter=True,
           max_retries=5,
       )
       def canvas_sync_user(self, user_id: str, ical_url: str) -> dict:
           """Fetch, parse, diff, and persist one user's Canvas deadlines.

           Uses exponential backoff on HTTP errors (including 429 rate limit).
           synced_at is only written after apply_diff succeeds.
           """
           from canvas.fetch import fetch_ical
           from canvas.parse import parse_vevents
           from canvas.diff import compute_diff
           from canvas.db import load_current_deadlines, apply_diff, mark_synced

           raw = fetch_ical(ical_url)
           events = parse_vevents(raw)
           current = load_current_deadlines(user_id)
           diff = compute_diff(current, events)
           apply_diff(user_id, diff)
           mark_synced(user_id)
           return {
               "user_id": user_id,
               "added": len(diff["added"]),
               "changed": len(diff["changed"]),
               "removed": len(diff["removed"]),
           }
       ```
