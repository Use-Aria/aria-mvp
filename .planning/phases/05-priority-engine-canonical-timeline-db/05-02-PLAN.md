    -- Phase 5: Canonical Timeline DB — unified event store for all integrations.
    -- Canvas deadlines are backfilled into this table via Celery task on deploy.
    -- IMPORTANT: Never run this migration without notifying the other developer first.

    CREATE TABLE public.canonical_events (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      user_id UUID NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
      source TEXT NOT NULL CHECK (source IN ('canvas', 'gcal', 'gmail', 'todoist')),
      type TEXT,
      title TEXT NOT NULL,
      due_at TIMESTAMPTZ NOT NULL,
      course_id TEXT,
      grade_weight NUMERIC,
      priority_score INTEGER,
      delivery_status TEXT NOT NULL DEFAULT 'silent'
        CHECK (delivery_status IN ('immediate', 'digest', 'silent', 'sent')),
      snoozed_until TIMESTAMPTZ,
      deleted_at TIMESTAMPTZ,
      created_at TIMESTAMPTZ DEFAULT now(),
      UNIQUE (user_id, source, title, due_at)
    );

    ALTER TABLE public.canonical_events ENABLE ROW LEVEL SECURITY;

    CREATE POLICY "User-scoped" ON public.canonical_events FOR ALL TO authenticated
      USING ((SELECT auth.uid()) = user_id)
      WITH CHECK ((SELECT auth.uid()) = user_id);

    CREATE INDEX idx_canonical_events_briefing
      ON public.canonical_events (user_id, due_at)
      WHERE deleted_at IS NULL
        AND delivery_status IN ('digest', 'immediate');

    CREATE INDEX idx_canonical_events_delivery
      ON public.canonical_events (user_id, delivery_status, created_at)
      WHERE delivery_status = 'immediate';
    ```

    def send_sms_alert(user_id: str, to_phone: str, body: str) -> bool:
        """Send time-sensitive alert SMS bypassing quiet hours and daily cap.

        Still checks opt-out — never send to opted-out users regardless of score.
        Reserved for priority_score >= 80 events only.
        Requires explicit onboarding consent before Yale beta (TCPA — Phase 8 Seb task).

        Args:
            user_id: UUID of recipient (used for opt-out check and hashed logging).
            to_phone: Recipient E.164 phone number. Never logged in plaintext.
            body: Alert message body.

        Returns:
            True if sent, False if user is opted out.
        """
        user = _get_user_sms_state(user_id)
        hashed_id = hashlib.sha256(user_id.encode()).hexdigest()[:12]

        if user["sms_opted_out"]:
            _log_suppressed(hashed_id, "opted_out")
            return False

        account_sid = os.environ["TWILIO_ACCOUNT_SID"]
        auth_token = os.environ["TWILIO_AUTH_TOKEN"]
        from_number = os.environ["TWILIO_PHONE_NUMBER"]
        client = Client(account_sid, auth_token)
        message = client.messages.create(from_=from_number, to=to_phone, body=body)
        logger.info(json.dumps({
            "event": "sms_alert_sent",
            "sid": message.sid,
            "uid": hashed_id,
        }))
        return True
    ```

    **Step 1: Add send_priority_alert task to tasks.py**
    Add a new Celery task:
    ```python
    @celery_app.task(
        name="tasks.send_priority_alert",
        bind=True,
        autoretry_for=(Exception,),
        retry_backoff=2,
        max_retries=3,
    )
    def send_priority_alert(self, user_id: str, event_id: str) -> dict:
        """Send immediate SMS alert for a high-priority canonical event.

        Fetches the event from canonical_events, builds alert message body,
        sends via send_sms_alert(), and updates delivery_status to 'sent'.
        """
        from ai.db import _get_supabase
        from sms.send import send_sms_alert as _send_alert, _get_user_sms_state

        client = _get_supabase()
        event = (
            client.table("canonical_events")
            .select("title, due_at, course_id, type, priority_score")
            .eq("id", event_id)
            .single()
            .execute()
        ).data

        user = _get_user_sms_state(user_id)
        to_phone = user.get("phone", "")
        if not to_phone:
            return {"user_id": user_id, "status": "error", "reason": "no_phone"}

        # Build alert body
        course = event.get("course_id") or "Unknown course"
        title = event["title"]
        due = event["due_at"]
        body = f"URGENT: {title} ({course}) is due soon — {due}. Priority score: {event['priority_score']}/100."

        sent = _send_alert(user_id, to_phone, body)

        if sent:
            from ai.db import write_outbound_message
            write_outbound_message(user_id, body, "priority_alert")
            # Mark as sent
            client.table("canonical_events").update(
                {"delivery_status": "sent"}
            ).eq("id", event_id).execute()

        return {"user_id": user_id, "event_id": event_id, "status": "sent" if sent else "suppressed"}
    ```

        """One-shot: fan out backfill per user. Run once on Phase 5 deploy.

        from ai.db import _get_supabase

        result = client.table("deadlines").select("user_id").execute()
        user_ids = list(set(row["user_id"] for row in result.data))

    @celery_app.task(name="tasks.backfill_canonical_events_user")
    def backfill_canonical_events_user(user_id: str) -> dict:
        """Backfill one user's Canvas deadlines into canonical_events.

        Uses skip_routing=True to prevent spamming users with immediate alerts
        for already-known events.
        """
        from ai.db import _get_supabase
        from priority.ingest import ingest_canonical_event

        client = _get_supabase()
        result = (
            client.table("deadlines")
            .select("title, due_date, course, type, grade_weight")
            .eq("user_id", user_id)
            .is_("deleted_at", "null")
            .execute()

                event_type=d.get("type", "assignment"),

                due_at=d["due_date"],

                grade_weight=d.get("grade_weight"),
                skip_routing=True,
            )
            count += 1
        return {"user_id": user_id, "backfilled": count}
    ```

    def get_events_for_briefing(user_id: str, days_ahead: int = 7) -> list[dict]:
        """Fetch active digest/immediate events from canonical_events for briefing.

        Replaces get_deadlines_for_briefing() — reads canonical_events going forward.
        Returns events ordered by due_at ascending. Filters out:
        - deleted (deleted_at IS NOT NULL)
        - snoozed (snoozed_until > now, filtered in Python)
        - silent events (only digest and immediate included)

        After fetching, marks all returned digest events as 'sent' to prevent
        re-inclusion in future briefings.

        Args:
            user_id: UUID of the user.
            days_ahead: Number of days ahead to include events (default 7).

        Returns:
            List of event dicts with keys: id, title, due_at, course_id, type,
            grade_weight, priority_score, delivery_status, snoozed_until.
        """
        client = _get_supabase()
        now = datetime.now(timezone.utc)
        cutoff = (now + timedelta(days=days_ahead)).isoformat()
        result = (
            client.table("canonical_events")
            .select("id, title, due_at, course_id, type, grade_weight, priority_score, delivery_status, snoozed_until")
            .eq("user_id", user_id)
            .is_("deleted_at", "null")
            .in_("delivery_status", ["digest", "immediate"])
            .lte("due_at", cutoff)
            .order("due_at", desc=False)
            .execute()
        )
        active = []
        for e in result.data:
            snoozed = e.get("snoozed_until")
            if snoozed:
                snooze_dt = datetime.fromisoformat(str(snoozed))
                if now < snooze_dt:
                    continue
            active.append(e)

        # Mark digest events as 'sent' to prevent re-inclusion
        digest_ids = [e["id"] for e in active if e.get("delivery_status") == "digest"]
        if digest_ids:
            for eid in digest_ids:
                client.table("canonical_events").update(
                    {"delivery_status": "sent"}
                ).eq("id", eid).execute()

        return active
    ```
