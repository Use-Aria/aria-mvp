    4. On mount (useEffect), generate nonce:
    ```typescript
    useEffect(() => {
      const generateNonce = async () => {
        const nonce = Crypto.randomUUID()
        setRawNonce(nonce)
        const hashed = await Crypto.digestStringAsync(
          Crypto.CryptoDigestAlgorithm.SHA256,
          nonce
        )
        setHashedNonce(hashed)
      }
      generateNonce()
    }, [])
    ```

    5. Google auth request hook:
    ```typescript
    const [request, response, promptAsync] = Google.useAuthRequest({
      iosClientId: process.env.EXPO_PUBLIC_GOOGLE_IOS_CLIENT_ID,
      webClientId: process.env.EXPO_PUBLIC_GOOGLE_WEB_CLIENT_ID,
      scopes: ['https://www.googleapis.com/auth/calendar', 'https://www.googleapis.com/auth/gmail.readonly'],
      extraParams: hashedNonce ? { nonce: hashedNonce } : undefined,
    })
    ```

    6. Handle response (useEffect on response):
    ```typescript
    useEffect(() => {
      if (response?.type === 'success') {
        setLoading(true)
        const { id_token } = response.params
        supabase.auth.signInWithIdToken({
          provider: 'google',
          token: id_token,
          nonce: rawNonce, // raw nonce, NOT hashed
        }).then(({ error }) => {
          setLoading(false)
          if (error) {
            setError("Couldn't connect Google. Try again or skip for now.")
          } else {
            router.push('/(onboarding)/tcpa')
          }
        })
      } else if (response?.type === 'error') {
        setError("Couldn't connect Google. Try again or skip for now.")
      }
    }, [response])
    ```

    ```typescript
    const handleConfirm = async () => {
      setSaving(true)
      const { data: { session } } = await supabase.auth.getSession()
      if (!session) return

      // Read current preferences first
      const { data: user } = await supabase
        .from('users')
        .select('preferences_json')
        .eq('id', session.user.id)
        .single()

      const currentPrefs = user?.preferences_json ?? {}

      const { error } = await supabase
        .from('users')
        .update({
          preferences_json: { ...currentPrefs, has_alert_consent: hasConsent }
        })
        .eq('id', session.user.id)

      setSaving(false)
      if (!error) {
        router.push('/(onboarding)/preview')
      }
    }
    ```

    Note: `has_alert_consent` is written as explicit `false` if toggle is OFF (not omitted). This is intentional — the flag is read by Phase 5's `send_sms_alert()`.
