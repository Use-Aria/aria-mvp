    Create `celery_worker/todoist/sync.py`:
    ```python
    import requests
    from todoist.fetch import get_active_tasks
    from todoist.parse import parse_tasks
    from todoist.db import get_todoist_token, mark_synced, mark_integration_revoked
    from priority.ingest import ingest_canonical_event

    def sync_todoist_user(user_id: str) -> dict:
        access_token = get_todoist_token(user_id)
        if not access_token:
            return {"user_id": user_id, "status": "no_token"}
        try:
            raw_tasks = get_active_tasks(access_token)
        except requests.HTTPError as e:
            if e.response is not None and e.response.status_code == 403:
                mark_integration_revoked(user_id)
                return {"user_id": user_id, "status": "token_revoked"}
            raise  # re-raise for Celery retry
        parsed = parse_tasks(raw_tasks)
        for t in parsed:
            ingest_canonical_event(
                user_id=user_id,
                source="todoist",
                event_type="task",
                title=t["title"],
                due_at=t["due_date"],
                course_id=None,
                grade_weight=0.0,
            )
        mark_synced(user_id)
        return {"user_id": user_id, "synced": len(parsed)}
    ```

    IMPORTANT SECURITY: `todoist_sync_all` passes only `user_id` to `todoist_sync_user.delay(user_id)` — NOT the access token. The token is fetched from Supabase inside `sync_todoist_user()` to keep tokens out of Redis.

    Modify `celery_worker/tasks.py` — add two new tasks after the existing `backfill_canonical_events_user` task:

    ```python
    @celery_app.task(name="tasks.todoist_sync_all")
    def todoist_sync_all():
        from todoist.db import get_all_todoist_integrations
        integrations = get_all_todoist_integrations()
        for row in integrations:
            todoist_sync_user.delay(row["user_id"])
        return {"status": "ok", "dispatched": len(integrations)}

    @celery_app.task(
        name="tasks.todoist_sync_user",
        bind=True,
        autoretry_for=(Exception,),
        retry_backoff=2,
        retry_backoff_max=300,
        retry_jitter=True,
        max_retries=5,
    )
    def todoist_sync_user(self, user_id: str) -> dict:
        from todoist.sync import sync_todoist_user
        return sync_todoist_user(user_id)
    ```

    Note on retry: The `autoretry_for=(Exception,)` will handle transient HTTP errors. The 403 case is caught inside `sync_todoist_user()` which returns (does not raise), so Celery will not retry on revoked tokens.
